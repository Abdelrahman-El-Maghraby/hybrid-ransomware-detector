# All dependencies are pure-Python (architecture-independent) or ship prebuilt
# wheels for x86_64 AND ARM64 on Windows, macOS, and Linux — so pip installs the
# correct build automatically (incl. Apple Silicon). No CPU-arch pinning; no
# source compilation required.

# Core
requests
schedule
python-dotenv

# Document parsing
PyPDF2
python-docx
pdf2image           # also needs the system package "poppler"
pytesseract         # also needs the system package "tesseract-ocr"
Pillow              # ships x86_64 + arm64 wheels for all major OSes

# Graphical interface (gui.py / --gui)
customtkinter       # pure-Python; needs Tkinter (bundled on Windows/macOS;
                    # on Debian/Ubuntu: sudo apt-get install python3-tk)

# --- Detection layers (all optional; the scanner degrades gracefully if any
#     are missing, so a minimal install still runs the hash + DB layers) -------

# Fuzzy / similarity hashing (polymorphic family detection)
python-tlsh         # TLSH locality-sensitive hash; prebuilt wheels x86_64+arm64
ppdeep              # pure-Python ssdeep; no libfuzzy/compiler needed

# VBA / XLM macro analysis for Office documents (dropper detection)
oletools            # pure-Python; provides olevba

# YARA structural pattern matching (IOCs)
yara-python         # ships prebuilt wheels for x86_64 + arm64 on all major OSes

# --- v4 layers & protection features ------------------------------------------
# Safe XML parsing for hostile Office files (hidden-content + external-link
# layers). Pure-Python. Without it those two layers fall back to PDF/RTF only.
defusedxml
# AES-256-GCM quarantine vault. Prebuilt wheels for x86_64 + arm64 everywhere.
# Without it scanning still works; only quarantine is unavailable.
cryptography

# Tests (optional):  pip install pytest qrcode   then   python -m pytest

# Local LLM backend for OPSEC (no extra Python dep — talks to a running Ollama
# server over HTTP via 'requests', already listed above). Install Ollama itself
# from https://ollama.com and pull a model, e.g.:  ollama pull llama3.1

# Windows-only (optional — only needed for the --create-shortcut .lnk feature)
# pywin32 ; sys_platform == "win32"

# --- v4.1 ----------------------------------------------------------------------
# .msg (Outlook) e-mails are read with olefile, which oletools already installs.
# ML classifier training/model loading:  pip install -r requirements-ml.txt
# Standalone executables:               pip install -r requirements-build.txt

# --- v4.2 (all optional; without them the feature reports what to install) ----
pyzipper            # AES-encrypted (WinZip) ZIP members. Pure-Python.
py7zr               # .7z archives, including encrypted file names. Pure-Python.
pycdlib             # .iso / .img disc images (ISO 9660, Joliet, Rock Ridge). Pure-Python.
watchdog            # real-time canary watching; prebuilt wheels x86_64 + arm64.

# --- v4.3 ----------------------------------------------------------------------
rarfile             # .rar archives. Pure-Python reader; decompression also needs
                    # one program already on the machine: 7-Zip (Windows: found in
                    # Program Files automatically), unrar, unar or bsdtar.
                    # Linux: sudo apt-get install 7zip   macOS: brew install sevenzip

# --- v4.4 (optional) -------------------------------------------------------------
pyzbar              # QR codes in images / PDFs / Office files (quishing). Windows
                    # wheels include zbar; Linux: sudo apt-get install libzbar0
                    # macOS: brew install zbar. Alternative decoder:
                    # pip install opencv-python-headless
